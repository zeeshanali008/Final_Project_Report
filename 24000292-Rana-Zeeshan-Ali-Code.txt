CarAgent Script for Robot Control

using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;
using Unity.MLAgents;
using UnityEditor;
using Unity.MLAgents.Sensors;
using Unity.MLAgents.Actuators;


public class CarAgent : Agent
{
    [Header("Wheel Colliders")]
    [SerializeField] private WheelCollider WheelColliderFrontLeft;
    [SerializeField] private WheelCollider WheelColliderFrontRight;
    [SerializeField] private WheelCollider WheelColliderBackLeft;
    [SerializeField] private WheelCollider WheelColliderBackRight;

    [Header("Wheel Transforms")]
    [SerializeField] private Transform TransformWheelFrontLeft;
    [SerializeField] private Transform TransformWheelFrontRight;
    [SerializeField] private Transform TransformWheelBackLeft;
    [SerializeField] private Transform TransformWheelBackRight;

    [Header("Sensors")]
    [SerializeField] private Transform RaycastSensorFront;
    [SerializeField] private Transform RaycastSensorLeft;
    [SerializeField] private Transform RaycastSensorRight;

    [Header("Checkpoints")]
    [SerializeField] private Transform[] checkpoints;
    [SerializeField] private Transform startPosition;

    [Header("Car Settings")]
    [SerializeField] private float engineForce = 600f;
    [SerializeField] private float steerLimit = 40.0f;
    [SerializeField] private float maxSpeed = 15f;
    [SerializeField] private float raycastDistance = 40f;

    [Header(" Generalization Settings")]
    [SerializeField] private bool enableRandomization = true;
    [SerializeField] private float engineForceVariation = 100f;  // Engine power randomize
    [SerializeField] private float frictionVariation = 0.3f;     // Track grip randomize
    [SerializeField] private float massVariation = 50f;          // Car weight randomize

    [Header("Advanced Raycasts")]
    [SerializeField] private bool useAdvancedRaycasts = true;
    [SerializeField] private int numberOfRays = 9;  // Zyada rays = better generalization

    [Header("Collision Log")]
    [SerializeField] private string algorithmName = "SAC";

    private Rigidbody rb;
    private float activeSteerAngle = 0f;
    private int currentCheckpointIndex = 0;
    private float episodeTimer = 0f;
    private float maxEpisodeTime = 90f;
    private Vector3 startPos;
    private Quaternion startRot;
    private float lastDistanceToCheckpoint;
    private int episodeCount = 0;
    private int collisionCount = 0;
    private int totalCollisions = 0;
    private string csvPath;
    private float episodeStartTime;
    private float trainingStartTime;

    // Randomization values
    private float currentEngineForce;
    private float originalMass;
    private WheelFrictionCurve originalForwardFriction;
    private WheelFrictionCurve originalSidewaysFriction;

    private void Awake()
    {
        rb = GetComponent<Rigidbody>();
        startPos = startPosition != null ? startPosition.position : transform.position;
        startRot = startPosition != null ? startPosition.rotation : transform.rotation;

        // Original values save karo
        originalMass = rb.mass;
        if (WheelColliderFrontLeft != null)
        {
            originalForwardFriction = WheelColliderFrontLeft.forwardFriction;
            originalSidewaysFriction = WheelColliderFrontLeft.sidewaysFriction;
        }

        // CSV setup
        string desktop = System.Environment.GetFolderPath(System.Environment.SpecialFolder.Desktop);
        csvPath = Path.Combine(desktop, algorithmName + "_collision_log.csv");
        if (!File.Exists(csvPath))
            File.WriteAllText(csvPath, "Episode,CollisionsThisEpisode,TotalCollisions,Steps,TimeTaken\n");

        trainingStartTime = Time.realtimeSinceStartup;
    }

    public override void OnEpisodeBegin()
    {
        episodeCount++;
        collisionCount = 0;
        episodeStartTime = Time.time;

        // RANDOMIZATION: Har episode mein thoda variation
        if (enableRandomization)
        {
            ApplyRandomization();
        }
        else
        {
            currentEngineForce = engineForce;
        }

        // Position randomize (thoda sa)
        Vector3 randomOffset = new Vector3(
            Random.Range(-0.5f, 0.5f),
            0,
            Random.Range(-0.5f, 0.5f)
        );
        transform.position = startPos + randomOffset;

        // Rotation bhi thoda randomize
        Quaternion randomRot = Quaternion.Euler(
            0,
            Random.Range(-10f, 10f),  // -10 se +10 degree
            0
        );
        transform.rotation = startRot * randomRot;

        rb.velocity = Vector3.zero;
        rb.angularVelocity = Vector3.zero;
        activeSteerAngle = 0f;
        currentCheckpointIndex = 0;
        episodeTimer = 0f;

        WheelColliderFrontLeft.brakeTorque = 0;
        WheelColliderFrontRight.brakeTorque = 0;
        WheelColliderBackLeft.brakeTorque = 0;
        WheelColliderBackRight.brakeTorque = 0;

        if (checkpoints != null && checkpoints.Length > 0)
            lastDistanceToCheckpoint = Vector3.Distance(transform.position, checkpoints[0].position);
    }


    private void ApplyRandomization()
    {
        // 1. Engine Force Variation
        currentEngineForce = engineForce + Random.Range(-engineForceVariation, engineForceVariation);

        // 2. Mass Variation (car ki weight)
        rb.mass = originalMass + Random.Range(-massVariation, massVariation);

        // 3. Friction Variation (track grip)
        WheelFrictionCurve forward = originalForwardFriction;
        WheelFrictionCurve sideways = originalSidewaysFriction;

        float frictionMultiplier = 1.0f + Random.Range(-frictionVariation, frictionVariation);
        forward.stiffness = originalForwardFriction.stiffness * frictionMultiplier;
        sideways.stiffness = originalSidewaysFriction.stiffness * frictionMultiplier;

        WheelColliderFrontLeft.forwardFriction = forward;
        WheelColliderFrontLeft.sidewaysFriction = sideways;
        WheelColliderFrontRight.forwardFriction = forward;
        WheelColliderFrontRight.sidewaysFriction = sideways;
        WheelColliderBackLeft.forwardFriction = forward;
        WheelColliderBackLeft.sidewaysFriction = sideways;
        WheelColliderBackRight.forwardFriction = forward;
        WheelColliderBackRight.sidewaysFriction = sideways;
    }

    public override void CollectObservations(VectorSensor sensor)
    {
        // 1. Basic car state
        sensor.AddObservation(rb.velocity.magnitude / maxSpeed);
        sensor.AddObservation(transform.forward);
        sensor.AddObservation(transform.right);

        // 2. Angular velocity (turning speed)
        sensor.AddObservation(rb.angularVelocity.y / 5f);        // Normalized turning


        if (useAdvancedRaycasts)
        {
            // Front rays (5 rays: -45, -22.5, 0, 22.5, 45 degrees)
            for (int i = 0; i < 5; i++)
            {
                float angle = -45f + (i * 22.5f);
                sensor.AddObservation(GetRaycastDistance(RaycastSensorFront, angle));
            }

            // Side rays
            sensor.AddObservation(GetRaycastDistance(RaycastSensorLeft, -90f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorLeft, -60f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorRight, 90f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorRight, 60f));
        }
        else
        {
            // Simple raycasts (original)
            sensor.AddObservation(GetRaycastDistance(RaycastSensorFront, 0f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorFront, -25f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorFront, 25f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorLeft, -50f));
            sensor.AddObservation(GetRaycastDistance(RaycastSensorRight, 50f));
        }

        // 4. Checkpoint information
        if (checkpoints != null && currentCheckpointIndex < checkpoints.Length)
        {
            Vector3 toCheckpoint = checkpoints[currentCheckpointIndex].position - transform.position;
            Vector3 dirToCheckpoint = toCheckpoint.normalized;

            sensor.AddObservation(dirToCheckpoint);                              // Direction to checkpoint (3 values)
            sensor.AddObservation(Vector3.Dot(transform.forward, dirToCheckpoint)); // How aligned we are
            sensor.AddObservation(toCheckpoint.magnitude / 50f);                 // Distance (normalized)
        }
        else
        {
            sensor.AddObservation(Vector3.zero);  // 3 values
            sensor.AddObservation(0f);
            sensor.AddObservation(0f);
        }

        // 5. Current steering angle (helps smooth steering)
        sensor.AddObservation(activeSteerAngle / steerLimit);
    }

    public override void OnActionReceived(ActionBuffers actions)
    {
        episodeTimer += Time.fixedDeltaTime;
        float throttle = Mathf.Clamp(actions.ContinuousActions[0], -1f, 1f);
        float steering = Mathf.Clamp(actions.ContinuousActions[1], -1f, 1f);

        // Movement Logic
        float targetSteer = steering * steerLimit;
        activeSteerAngle = Mathf.Lerp(activeSteerAngle, targetSteer, Time.fixedDeltaTime * 5f);
        WheelColliderFrontLeft.steerAngle = activeSteerAngle;
        WheelColliderFrontRight.steerAngle = activeSteerAngle;

        if (throttle > 0)
        {
            float torque = throttle * currentEngineForce;  // Use randomized engine force
            WheelColliderFrontLeft.motorTorque = torque;
            WheelColliderFrontRight.motorTorque = torque;
            WheelColliderBackLeft.motorTorque = torque;
            WheelColliderBackRight.motorTorque = torque;
            SetBrakes(0);
        }
        else
        {
            SetMotorTorque(0);
            SetBrakes(Mathf.Abs(throttle) * 1000f);
        }

        SyncWheelVisuals();


        float forwardSpeed = Vector3.Dot(rb.velocity, transform.forward);

        // 1. Speed Reward - Forward movement ko reward do
        if (forwardSpeed > 0.5f)
        {
            AddReward(0.08f * (forwardSpeed / maxSpeed));
        }
        else if (forwardSpeed < -0.5f)
        {
            // Backward movement penalty
            AddReward(-0.02f);
        }

        // 2. Checkpoint Proximity Reward
        if (checkpoints != null && currentCheckpointIndex < checkpoints.Length)
        {
            float dist = Vector3.Distance(transform.position, checkpoints[currentCheckpointIndex].position);
            float delta = lastDistanceToCheckpoint - dist;

            if (delta > 0f)
            {
                // Checkpoint ke paas ja rahe hain
                AddReward(delta * 0.03f);
            }
            else if (delta < -0.1f)
            {
                // Checkpoint se door ja rahe hain
                AddReward(-0.005f);
            }

            lastDistanceToCheckpoint = dist;
        }

        // 3. Steering smoothness reward
        float steeringChange = Mathf.Abs(steering - (activeSteerAngle / steerLimit));
        if (steeringChange < 0.1f)
        {
            AddReward(0.001f);  // Smooth steering ko reward
        }

        // 4. Small time penalty
        AddReward(-0.0001f);

        if (episodeTimer >= maxEpisodeTime)
        {
            AddReward(-0.5f);  // Time limit penalty
            EndEpisode();
        }
    }

    private void SetBrakes(float force)
    {
        WheelColliderFrontLeft.brakeTorque = force;
        WheelColliderFrontRight.brakeTorque = force;
        WheelColliderBackLeft.brakeTorque = force;
        WheelColliderBackRight.brakeTorque = force;
    }

    private void SetMotorTorque(float torque)
    {
        WheelColliderFrontLeft.motorTorque = torque;
        WheelColliderFrontRight.motorTorque = torque;
        WheelColliderBackLeft.motorTorque = torque;
        WheelColliderBackRight.motorTorque = torque;
    }

    public override void Heuristic(in ActionBuffers actionsOut)
    {
        var ca = actionsOut.ContinuousActions;
        ca[0] = Input.GetAxis("Vertical");
        ca[1] = Input.GetAxis("Horizontal");
    }

    private void OnCollisionEnter(Collision collision)
    {
        if (collision.gameObject.CompareTag("Obstacle") ||
            collision.gameObject.layer == LayerMask.NameToLayer("Obs"))
        {
            collisionCount++;
            totalCollisions++;

            float timeTaken = Time.time - episodeStartTime;
            File.AppendAllText(csvPath,
                $"{episodeCount},{collisionCount},{totalCollisions},{StepCount},{timeTaken:F2}\n");

            // Big penalty for collision
            SetReward(-1.0f);
            EndEpisode();
        }
    }

    private void OnTriggerEnter(Collider other)
    {
        if (!other.CompareTag("Checkpoint")) return;

        CheckpointScript cp = other.GetComponent<CheckpointScript>();
        if (cp == null) return;

        if (cp.index == currentCheckpointIndex)
        {
            // Correct checkpoint hit!
            AddReward(2.0f);  // Increased reward
            currentCheckpointIndex++;

            if (currentCheckpointIndex >= checkpoints.Length)
            {
                
                AddReward(15.0f);  // Big completion reward

                float timeTaken = Time.time - episodeStartTime;
                File.AppendAllText(csvPath,
                    $"{episodeCount},{collisionCount},{totalCollisions},{StepCount},{timeTaken:F2}\n");

                float totalTrainingTime = Time.realtimeSinceStartup - trainingStartTime;
                string timePath = Path.Combine(
                    System.Environment.GetFolderPath(System.Environment.SpecialFolder.Desktop),
                    algorithmName + "_training_time.txt"
                );
                File.WriteAllText(timePath, totalTrainingTime.ToString("F2"));

                Debug.Log($" TRAINING COMPLETE! Total time: {totalTrainingTime:F2}s");

                EndEpisode();

#if UNITY_EDITOR
                EditorApplication.isPlaying = false;
#endif
            }
            else
            {
                lastDistanceToCheckpoint = Vector3.Distance(
                    transform.position,
                    checkpoints[currentCheckpointIndex].position
                );
            }
        }
        else if (cp.index > currentCheckpointIndex)
        {
            // Wrong checkpoint order
            AddReward(-0.2f);
        }
    }

    private float GetRaycastDistance(Transform sensor, float angle)
    {
        if (sensor == null) return 1f;

        Vector3 dir = Quaternion.Euler(0, angle, 0) * sensor.forward;
        RaycastHit hit;

        if (Physics.Raycast(sensor.position, dir, out hit, raycastDistance))
        {
            // Normalized distance
            return hit.distance / raycastDistance;
        }

        return 1f;  // No obstacle detected
    }

    private void SyncWheelVisuals()
    {
        UpdateWheelPositioning(WheelColliderFrontLeft, TransformWheelFrontLeft);
        UpdateWheelPositioning(WheelColliderFrontRight, TransformWheelFrontRight);
        UpdateWheelPositioning(WheelColliderBackLeft, TransformWheelBackLeft);
        UpdateWheelPositioning(WheelColliderBackRight, TransformWheelBackRight);
    }

    private void UpdateWheelPositioning(WheelCollider col, Transform mdl)
    {
        if (col == null || mdl == null) return;
        col.GetWorldPose(out Vector3 pos, out Quaternion rot);
        mdl.position = pos;
        mdl.rotation = rot;
    }

    private void OnDrawGizmos()
    {
        if (!Application.isPlaying || RaycastSensorFront == null) return;

        // Draw raycasts in editor
        Gizmos.color = Color.yellow;
        for (int i = 0; i < 5; i++)
        {
            float angle = -45f + (i * 22.5f);
            Vector3 dir = Quaternion.Euler(0, angle, 0) * RaycastSensorFront.forward;
            Gizmos.DrawRay(RaycastSensorFront.position, dir * raycastDistance);
        }
    }
}


CheckpointScript for Count CheckPoints

using UnityEngine;


public class CheckpointScript : MonoBehaviour
{
    [Header("Checkpoint Settings")]
    public int index = 0;  

    [Header("Visual Settings")]
    public Color checkpointColor = Color.green;
    public bool showInGame = false;  

    [Header("Debug Info")]
    public int timesHit = 0;  

    private void OnTriggerEnter(Collider other)
    {
        
        if (other.CompareTag("Player") || other.GetComponent<CarAgent>() != null)
        {
            timesHit++;
            Debug.Log($" Checkpoint {index} hit! Total hits: {timesHit}");
        }
    }

    private void OnDrawGizmos()
    {
        
        Gizmos.color = new Color(checkpointColor.r, checkpointColor.g, checkpointColor.b, 0.3f);
        Gizmos.DrawCube(transform.position, transform.localScale);

        Gizmos.color = checkpointColor;
        Gizmos.DrawWireCube(transform.position, transform.localScale);

        // Checkpoint index show karo
#if UNITY_EDITOR
        UnityEditor.Handles.Label(
            transform.position + Vector3.up * 2f,
            $"CP {index}",
            new GUIStyle()
            {
                normal = new GUIStyleState() { textColor = Color.white },
                fontSize = 14,
                fontStyle = FontStyle.Bold
            }
        );
#endif
    }

    private void OnDrawGizmosSelected()
    {
        // Selected ho to direction arrow dikhao
        Gizmos.color = Color.cyan;
        Gizmos.DrawRay(transform.position, transform.forward * 5f);
    }

    // Runtime mein checkpoint dikhane ke liye (optional)
    private MeshRenderer meshRenderer;

    private void Start()
    {
        meshRenderer = GetComponent<MeshRenderer>();
        if (meshRenderer != null)
        {
            meshRenderer.enabled = showInGame;
        }
    }

    // Inspector mein color change hone pe update karo
    private void OnValidate()
    {
        if (meshRenderer != null)
        {
            meshRenderer.enabled = showInGame;
        }
    }
}
